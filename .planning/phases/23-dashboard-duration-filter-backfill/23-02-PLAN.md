    loadDurationBounds(): void {
      this.loadingDurationBounds = true;
      const params: any = {
        date_from: this.dateFrom,
        date_to: this.dateTo,
      };
      if (this.selectedPlatforms?.size) {
        params.platforms = [...this.selectedPlatforms].join(',');
      }
      if (this.selectedFormat) params.formats = this.selectedFormat;
      if (this.selectedAdAccountIds.length > 0) params.ad_account_ids = this.selectedAdAccountIds.join(',');
      if (this.activeMetadataFilters.length > 0) {
        params.metadata_filter = this.activeMetadataFilters.map(f => `${f.field}:${f.value}`);
      }
      // NOTE: do NOT pass duration_min / duration_max — that would be circular (D-02)

      this.api.get<{ min_duration: number; max_duration: number }>('/dashboard/duration-bounds', params).subscribe({
        next: (res) => {
          this.durationSliderOptions = {
            ...this.durationSliderOptions,
            floor: res.min_duration,
            ceil: res.max_duration,
          };
          // Reset handles to full range (avoids stuck state when bounds shift)
          this.durationMin = res.min_duration;
          this.durationMax = res.max_duration;
          this.loadingDurationBounds = false;
        },
        error: () => {
          this.durationSliderOptions = { ...this.durationSliderOptions, floor: 0, ceil: 3600 };
          this.loadingDurationBounds = false;
        },
      });
    }

    clearDurationFilter(): void {
      this.durationMin = this.durationSliderOptions.floor ?? 0;
      this.durationMax = this.durationSliderOptions.ceil ?? 3600;
      this.onFilterChange();
    }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser → backend | duration_min/duration_max and the bounds endpoint are user-influenced; backend (Plan 01) enforces ge=0, org guards, and SQLAlchemy bind params. Frontend is a thin presentation layer over those guarantees. |
| component state → DOM | hasVideoAssets, nullDurationCount, durationMin/Max are reactive Angular state with no XSS surface (no innerHTML; values are bound via interpolation). |
| component → /dashboard/duration-bounds | The frontend passes only filter params; never tampers with org scoping. Org isolation enforced by JWT on backend. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-23-F-01 | Information Disclosure | Cached durationSliderOptions across orgs | mitigate | durationSliderOptions is component-instance state, cleared on navigation. loadDurationBounds() refetches whenever filters change (D-02) and on initial dashboard load. No localStorage persistence (URL persistence is explicitly deferred to v1.5). |
| T-23-F-02 | Tampering | User manually sets duration_min/max via URL query | mitigate | Phase 23 does not implement URL persistence (deferred to v1.5); even if a user crafts a URL with duration params, the backend (Plan 01 Task 4) enforces `ge=0` via FastAPI Query validation and org_id guard via JWT. Out-of-range or NaN values are rejected at the API boundary, not in the frontend. |
